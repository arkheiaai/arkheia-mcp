# Arkheia Profile Registry Server -- Python dependencies
#
# Previously these lived as a bare `RUN pip install fastapi uvicorn[standard]
# pyyaml httpx` literal in registry_server/Dockerfile, while the Dockerfile
# COPYed the root requirements.txt and never installed it. Two consequences:
# no dependency scanner (pip-audit / Snyk / Aikido all read requirements files)
# ever saw this image's dependency set, and the inline list drifted free of every
# CVE floor its sibling services pin. Declared here so the image's dependencies
# are scannable and pinned like everything else.
#
# Version floors follow the repo convention: `>=` on the fixed release, with the
# advisory that earned the floor named inline.

# Runtime imports: registry_server/main.py, registry_server/auth.py
fastapi>=0.135.1

# The container command: `uvicorn registry_server.main:app`
uvicorn[standard]>=0.31.1

# Runtime import: registry_server/storage.py (profile YAML load/dump)
pyyaml>=6.0

# Imported by the image HEALTHCHECK (`python -c "import httpx; ..."`), not by
# registry_server source -- so it is a real image dependency that no import-graph
# scan derives. Keep it declared.
httpx>=0.27.1

# Transitive via fastapi, pinned explicitly because the resolver would otherwise
# admit vulnerable releases (matches proxy/requirements.txt and requirements.txt).
starlette==1.3.1  # CVE-2025-62727 (Range-header DoS) + CVE-2025-54121 (multipart DoS) fixes
anyio>=4.14.2  # GHSA-82r6-8w77-94w6 (anyio IDNA-2008 TLS cert-validation) -- transitive pin
